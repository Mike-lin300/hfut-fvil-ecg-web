先说清楚我们对指针的要求。 指针是 C 语言里名声最吓人的部分，但我们对它的要求不高：理解概念、会用即可。 本节不讲二级指针、多级指针，不讲 const 与指针的各种组合，也不讲复杂的指针运算——那些东西很容易把初学者绕晕。 你只需要带走三样东西：指针存的是地址、***p 能顺着地址摸到那个变量、把地址传给函数就能改到原件**。

7.0 序

导学案2里我们说过一句话：变量 = 内存里一小片有名字的空间，地址是它的门牌号。 那时 &a 只是个"照做即可"的符号。这一节我们把它变成一件趁手的工具。

为什么绕不开它？因为单片机上的硬件，本质上就是一排排固定地址的格子（寄存器）。想让灯亮、让电机转，就是往某个地址写一个值。指针，就是 C 语言里"按地址操作内存"的唯一手段。

本节路线：地址与指针 → 指针在函数里怎么用 → 数组名本质是指针 → 练习

7.1 地址、指针、解引用

三个动作：

写法名字意思
&a取地址问出变量 a 住在几号门牌
int *p = &a;定义指针造一个专门存地址的变量 p，把 a 的门牌抄进去
*p解引用顺着 p 里存的门牌找过去，访问那个变量本身

int *p 表示 p 是一个指针，它指向一个 int。

#include <stdio.h>

int main(void)
{
	int a = 10;
	int *p = &a;             /* p 里存的是 a 的地址 */

	printf("a 的值      = %d\n", a);
	printf("a 的地址    = %p\n", (void *)&a);
	printf("p 里存的    = %p\n", (void *)p);
	printf("*p（解引用）= %d\n", *p);

	*p = 20;                 /* 顺着地址找到那个盒子，把值改掉 */
	printf("改完之后 a  = %d\n", a);

	return 0;
}

输出（地址每次运行、每台机器都不同，下面是其中一次）：

a 的值      = 10
a 的地址    = 00000075679FFC44
p 里存的    = 00000075679FFC44
*p（解引用）= 10
改完之后 a  = 20

几点说明：

  • %p 是打印地址的占位符。输出格式因编译器而异（有的打 0x0061FE1C，有的打 0061FE1C），认得出"这是一串十六进制的门牌号"就够了，不要纠结长相。
  • 最后一行是关键：我们没有碰 a 这个名字，只通过 *p 就把 a 改成了 20。***p 和 a 是同一个盒子的两种叫法。**
  • int p 里的 只在定义时表示"这是个指针"；定义之后，*p 才是解引用。同一个符号两个场合两种意思，和数组的 [5] 一样。

野指针警告：

/* 反面示例 */
int *p;          /* 没初始化，里面是个垃圾地址 */
*p = 5;          /* 往一个不知道是谁家的门牌里塞东西 */

这可能当场崩溃，也可能悄悄改掉别的变量。**规矩只有一条：定义指针时就给它一个地址；暂时没有，就写 int p = NULL;*（NULL 表示"不指向任何东西"）。

7.2 指针在函数里怎么用

终于能改到外面的变量了

回收导学案6 6.4 那个失败的 swap。这次传地址：

#include <stdio.h>

void swap(int *pa, int *pb)      /* 形参是两个指针：收到的是地址 */
{
	int t = *pa;
	*pa = *pb;
	*pb = t;
}

int main(void)
{
	int x = 1, y = 2;
	swap(&x, &y);                /* 把 x、y 的地址传进去 */
	printf("x = %d, y = %d\n", x, y);
	return 0;
}

输出：

x = 2, y = 1

这次真的换过来了。道理一句话：传值只能改副本，传地址就能改原件。 导学案2练习 5 的"三杯水"，到这里才算真正成功。

一个函数"带回"多个值

return 一次只能带回一个值。想带回两个怎么办？用指针把结果"写"到调用者准备好的盒子里：

#include <stdio.h>

void minMax(int a, int b, int *pmin, int *pmax)
{
	if (a < b) {
		*pmin = a;
		*pmax = b;
	} else {
		*pmin = b;
		*pmax = a;
	}
}

int main(void)
{
	int lo = 0, hi = 0;

	minMax(7, 3, &lo, &hi);
	printf("小的是 %d，大的是 %d\n", lo, hi);

	return 0;
}

输出：

小的是 3，大的是 7

这个模式你在飞控代码里会反复见到：LZY_Scheduler(..., &ModeChangedFlag) 就是把变量的地址交出去，让别的模块往里写结果。

顺便解开一个旧谜

导学案2让你死记"scanf 里每个变量前面都要写 &"。现在你懂了：

scanf("%d", &a) 就是把 a 的地址交给 scanf，让它顺着地址把读到的值放进 a。 如果不给地址，scanf 就不知道往哪放。

7.3 数组名本质是指针

这是本节第二个必须带走的结论：

数组名就是"第一个元素的地址"。 即 a 等价于 &a[0]。 因此 a[i] 等价于 *(a + i)：从首地址往后挪 i 个格子，再取里面的值。

#include <stdio.h>

int main(void)
{
	int a[5] = {10, 20, 30, 40, 50};
	int *p = a;                  /* 数组名直接赋给指针，不用写 & */

	printf("a[0] = %d, *p = %d\n", a[0], *p);
	printf("a[2] = %d, *(p + 2) = %d, p[2] = %d\n", a[2], *(p + 2), p[2]);
	printf("a == &a[0] 吗？ %d\n", a == &a[0]);

	return 0;
}

输出：

a[0] = 10, *p = 10
a[2] = 30, *(p + 2) = 30, p[2] = 30
a == &a[0] 吗？ 1

这个结论顺手解释了两件旧事：

  1. 为什么 scanf("%s", name) 不写 &：name 本身就是地址，再写 & 反而不对。
  2. 为什么数组传给函数不用复制整个数组：传一个地址就够了，又快又省。

数组作函数参数

#include <stdio.h>

void printArr(int a[], int len)        /* 也可以写成 int *a，两种写法完全等价 */
{
	for (int i = 0; i < len; i++) {
		printf("%d ", a[i]);
	}
	printf("\n");
}

void doubleAll(int a[], int len)
{
	for (int i = 0; i < len; i++) {
		a[i] *= 2;                     /* 传进来的是地址，所以能改到原数组 */
	}
}

int main(void)
{
	int score[5] = {90, 85, 77, 68, 99};
	int len = (int)(sizeof(score) / sizeof(score[0]));

	printArr(score, len);
	doubleAll(score, len);
	printArr(score, len);

	return 0;
}

输出：

90 85 77 68 99
180 170 154 136 198

注意第二行：doubleAll 里改的就是 main 里的那个数组。数组作参数时是"传地址"，不是"值传递"，这是它和普通 int 参数最大的区别。

一个必须记住的坑：长度要单独传。 函数里的 a 只是一个指针，sizeof(a) 得到的是指针本身的大小（4 或 8 字节），不是数组的大小。所以导学案5说"sizeof(a)/sizeof(a[0]) 这个套路只在数组定义的地方有效"——原因就在这里。

void swap(double *pa, double *pb)

练习 2 参考代码
#include <stdio.h>

void minMaxArr(int a[], int len, int *pmin, int *pmax)
{
	*pmin = a[0];
	*pmax = a[0];

	for (int i = 1; i < len; i++) {
		if (a[i] < *pmin) {
			*pmin = a[i];
		}
		if (a[i] > *pmax) {
			*pmax = a[i];
		}
	}
}

int main(void)
{
	int a[6] = {5, 2, 9, 1, 7, 3};
	int len = (int)(sizeof(a) / sizeof(a[0]));
	int lo = 0, hi = 0;

	minMaxArr(a, len, &lo, &hi);
	printf("最小 = %d，最大 = %d\n", lo, hi);

	return 0;
}

输出：最小 = 1，最大 = 9

练习 3 参考代码
#include <stdio.h>

void printArr(int a[], int len)
{
	for (int i = 0; i < len; i++) {
		printf("%d ", a[i]);
	}
	printf("\n");
}

void zeroEven(int a[], int len)
{
	for (int i = 0; i < len; i++) {
		if (a[i] % 2 == 0) {
			a[i] = 0;
		}
	}
}

int main(void)
{
	int a[6] = {4, 7, 2, 9, 6, 1};
	int len = (int)(sizeof(a) / sizeof(a[0]));

	printArr(a, len);
	zeroEven(a, len);
	printArr(a, len);

	return 0;
}

输出：

4 7 2 9 6 1
0 7 0 9 0 1

指针不可怕，可怕的是"不知道指针里存的是什么"。
